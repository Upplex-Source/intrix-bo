Contact-us API

1. Contact Us Form Submission

Example request:
curl --request GET \
    --get "http://localhost/api/v1/contact-us?full_name=Adam&email=adameve%40mail.com&phone_number=0123878123&location=Selangor&model=debitis" \
    --header "Content-Type: application/json" \
    --header "Accept: application/json"
const url = new URL(
    "http://localhost/api/v1/contact-us"
);

const params = {
    "full_name": "Adam",
    "email": "adameve@mail.com",
    "phone_number": "0123878123",
    "location": "Selangor",
    "model": "debitis",
};
Object.keys(params)
    .forEach(key => url.searchParams.append(key, params[key]));

const headers = {
    "Content-Type": "application/json",
    "Accept": "application/json",
};

fetch(url, {
    method: "GET",
    headers,
}).then(response => response.json());
